# ---- OCR ----
# paddlepaddle 3.2.2 is the newest release that still ships a
# manylinux2014_aarch64 wheel (3.3.x dropped it). That is what lets this one
# Dockerfile build natively on Apple Silicon AND on x86 for ECS Fargate.
#
# PaddleOCR 3.x pulls paddlex[ocr-core], which downloads model weights from
# Hugging Face by default (PADDLE_PDX_MODEL_SOURCE=huggingface) instead of
# Baidu's bcebos CDN -- bcebos is unreachable from this network, which is what
# broke the PaddleOCR 2.x build.
paddlepaddle==3.2.2
paddleocr==3.5.0
